cmake_minimum_required(VERSION 3.21)
project(BSIPortableLauncher LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The portable launcher requires Windows x64 and MSVC")
endif()
if(NOT EXISTS "${BSI_PAYLOAD_CAB}" OR NOT BSI_APP_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(\\.[0-9]+)?$")
    message(FATAL_ERROR "Supply BSI_PAYLOAD_CAB and a numeric BSI_APP_VERSION")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(SHA256 "${BSI_PAYLOAD_CAB}" BSI_PAYLOAD_SHA256)
file(TO_CMAKE_PATH "${BSI_PAYLOAD_CAB}" BSI_PAYLOAD_CAB)
get_filename_component(BSI_MANIFEST "../../src/app.manifest" ABSOLUTE)
file(TO_CMAKE_PATH "${BSI_MANIFEST}" BSI_MANIFEST)
string(REPLACE "." "," BSI_FILE_VERSION "${BSI_APP_VERSION}")
if(BSI_APP_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    string(APPEND BSI_FILE_VERSION ",0")
endif()
configure_file(launcher.rc.in launcher.rc @ONLY)
configure_file(payload.h.in payload.h @ONLY)
add_executable(BSIPortableLauncher WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
target_include_directories(BSIPortableLauncher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(BSIPortableLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(BSIPortableLauncher PRIVATE /W4 /utf-8)
target_link_libraries(BSIPortableLauncher PRIVATE cabinet bcrypt advapi32 shell32)
target_link_options(BSIPortableLauncher PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
set_property(TARGET BSIPortableLauncher APPEND PROPERTY VS_GLOBALS "GenerateManifest=false")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" PROPERTIES
    OBJECT_DEPENDS "${BSI_PAYLOAD_CAB};${BSI_MANIFEST}")
set_target_properties(BSIPortableLauncher PROPERTIES OUTPUT_NAME "BSI-Customizer-Portable")
