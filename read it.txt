# ---- PE parsing (no external tools) ----
function Parse-PE([byte[]]$b) {
    if ($b.Length -lt 0x100) { throw 'file too small / not a DLL' }
    if ($b[0] -ne 0x4D -or $b[1] -ne 0x5A) { throw 'not a Windows DLL/EXE (no MZ header)' }
    $lfanew = [BitConverter]::ToInt32($b, 0x3C)
    if ($lfanew -le 0 -or $lfanew -ge $b.Length - 24) { throw 'bad PE header' }
    if ($b[$lfanew] -ne 0x50 -or $b[$lfanew+1] -ne 0x45) { throw 'not a PE file' }
    $machine = [BitConverter]::ToUInt16($b, $lfanew + 4)
    if ($machine -ne 0x8664) { throw 'not a 64-bit DLL - unsupported' }
    $numSec  = [BitConverter]::ToUInt16($b, $lfanew + 6)
    $optSize = [BitConverter]::ToUInt16($b, $lfanew + 20)
    $opt     = $lfanew + 24
    if ([BitConverter]::ToUInt16($b, $opt) -ne 0x20B) { throw 'not PE32+ - unsupported' }
    $expRva = [BitConverter]::ToUInt32($b, $opt + 112)
    $secs = @()
    $secT = $opt + $optSize
    for ($i = 0; $i -lt $numSec; $i++) {
        $s = $secT + $i * 40
        if ($s + 40 -gt $b.Length) { break }
        $secs += [pscustomobject]@{
            Name   = [Text.Encoding]::ASCII.GetString($b, $s, 8).Trim([char]0)
            VA     = [BitConverter]::ToUInt32($b, $s + 12)
            VSz    = [BitConverter]::ToUInt32($b, $s + 8)
            RawSz  = [BitConverter]::ToUInt32($b, $s + 16)
            RawPtr = [BitConverter]::ToUInt32($b, $s + 20)
        }
    }
    return @{ ExpRva = $expRva; Sections = $secs }
}
function RvaToOff([int]$rva, $secs) {
    foreach ($s in $secs) {
        $span = [Math]::Max($s.VSz, $s.RawSz)
        if ($rva -ge $s.VA -and $rva -lt ($s.VA + $span)) { return $rva - $s.VA + $s.RawPtr }
    }
    throw ("RVA 0x{0:X} not mapped" -f $rva)
}
function Read-AsciiZ([byte[]]$b, [int]$off) {
    $e = $off
    while ($e -lt $b.Length -and $b[$e] -ne 0) { $e++ }
    return [Text.Encoding]::ASCII.GetString($b, $off, $e - $off)
}

# the slider functions we patch (MSVC + GCC name mangling)
$WANT = @{}
foreach ($cls in @('QQuickSlider', 'QQuickRangeSlider')) {
    $WANT['?setTo@' + $cls + '@@QEAAXN@Z'] = $cls
    $WANT['_ZN' + $cls.Length + $cls + '5setToEd'] = $cls
}
$PROLOGUE = @(0x48, 0x89, 0x5C, 0x24, 0x08)   # mov [rsp+8], rbx
$SLOT_LEN = 80
$CAVE_MIN = 2 * 80 + 16

function Find-Targets([byte[]]$b) {
    # returns @{ ok; state; msg; targets = @( @{name; cls; rva; off; state} ); caveOff; caveRva; textSec }
    $pe = Parse-PE $b
    if ($pe.ExpRva -eq 0) { return @{ ok = $false; state = 'unsupported'; msg = 'DLL has no export table' } }
    $ed = RvaToOff $pe.ExpRva $pe.Sections
    $nNames   = [BitConverter]::ToUInt32($b, $ed + 24)
    $funcsOff = RvaToOff ([BitConverter]::ToUInt32($b, $ed + 28)) $pe.Sections
    $namesOff = RvaToOff ([BitConverter]::ToUInt32($b, $ed + 32)) $pe.Sections
    $ordsOff  = RvaToOff ([BitConverter]::ToUInt32($b, $ed + 36)) $pe.Sections
    $targets = @()
    for ($i = 0; $i -lt $nNames; $i++) {
        $nRva = [BitConverter]::ToUInt32($b, $namesOff + $i * 4)
        $name = Read-AsciiZ $b (RvaToOff $nRva $pe.Sections)
        if ($WANT.ContainsKey($name)) {
            $ord  = [BitConverter]::ToUInt16($b, $ordsOff + $i * 2)
            $frva = [BitConverter]::ToUInt32($b, $funcsOff + $ord * 4)
            $off  = RvaToOff $frva $pe.Sections
            $st = 'unknown'
            $same = $true
            for ($k = 0; $k -lt 5; $k++) { if ($b[$off + $k] -ne $PROLOGUE[$k]) { $same = $false; break } }
            if ($same) { $st = 'clean' } elseif ($b[$off] -eq 0xE9) { $st = 'patched' }
            $targets += [pscustomobject]@{ Name = $name; Cls = $WANT[$name]; Rva = $frva; Off = $off; State = $st }
        }
    }
    if ($targets.Count -eq 0) {
        return @{ ok = $false; state = 'unsupported'; msg = 'slider functions (setTo) not found in exports - unsupported build' }
    }
    $anyClean   = (@($targets | Where-Object { $_.State -eq 'clean' }).Count -gt 0)
    $anyPatched = (@($targets | Where-Object { $_.State -eq 'patched' }).Count -gt 0)
    # already patched? detect that without needing a cave
    if ($anyPatched -and -not $anyClean) {
        return @{ ok = $false; state = 'patched'; msg = "$($targets.Count) setTo function(s), already patched"
                  Targets = $targets }
    }
    if (-not $anyClean) {
        return @{ ok = $false; state = 'unsupported'; msg = 'setTo found but unknown prologue - unsupported build'
                  Targets = $targets }
    }
    # find a code cave (only needed for patching): longest run of 00 in .text
    $text = $pe.Sections | Where-Object { $_.Name -eq '.text' } | Select-Object -First 1
    if (-not $text) { return @{ ok = $false; state = 'unsupported'; msg = 'no .text section' } }
    $best = -1; $bestLen = 0; $run = 0
    $i0 = $text.RawPtr; $i1 = $text.RawPtr + $text.RawSz
    for ($i = $i0; $i -lt $i1; $i++) {
        if ($b[$i] -eq 0) { $run++ } else {
            if ($run -gt $bestLen) { $bestLen = $run; $best = $i - $run }
            $run = 0
        }
    }
    if ($run -gt $bestLen) { $bestLen = $run; $best = $i1 - $run }
    if ($bestLen -lt $CAVE_MIN) {
        return @{ ok = $false; state = 'unsupported'; msg = "no empty padding area found (need $CAVE_MIN bytes)" }
    }
    $caveRva = $best - $text.RawPtr + $text.VA
    return @{ ok = $true; state = 'clean'
              msg = "$($targets.Count) setTo function(s) found; cave $($bestLen) bytes"
              Targets = $targets; CaveOff = $best; CaveRva = $caveRva }
}

function Build-Slot([int]$funcRva, [int]$caveRva, [double]$hi, [double]$lo) {
    $s = [System.Collections.Generic.List[byte]]::new()
    AB $s @(0x66,0x0F,0x2F,0x0D); AB $s (I32 (48 - 8))     # comisd xmm1,[c240]
    AB $s @(0x75, (20 - 10))                                # jne  -> +20
    AB $s @(0xF2,0x0F,0x10,0x0D); AB $s (I32 (64 - 18))    # movsd xmm1,[cHI]
    AB $s @(0xEB, (38 - 20))                                # jmp  -> +38
    AB $s @(0x66,0x0F,0x2F,0x0D); AB $s (I32 (56 - 28))    # comisd xmm1,[c60]
    AB $s @(0x75, (38 - 30))                                # jne  -> +38
    AB $s @(0xF2,0x0F,0x10,0x0D); AB $s (I32 (72 - 38))    # movsd xmm1,[cLO]
    AB $s @(0x48,0x89,0x5C,0x24,0x08)                       # mov [rsp+8],rbx
    AB $s @(0xE9); AB $s (I32 (($funcRva + 5) - ($caveRva + 48)))   # jmp back
    AB $s (D64 240.0); AB $s (D64 60.0); AB $s (D64 $hi); AB $s (D64 $lo)
    if ($s.Count -ne 80) { throw "internal slot size $($s.Count)" }
    return $s.ToArray()
}
function Build-Jmp([int]$fromRva, [int]$toRva) {
    $l = [System.Collections.Generic.List[byte]]::new()
    AB $l @(0xE9); AB $l (I32 ($toRva - ($fromRva + 5)))
    return $l.ToArray()
}

function Do-Patch([string]$dll, [double]$hi, [double]$lo) {
    $bak = "$dll.bak"
    Log "Target : $dll" 'Black'
    if (Test-Path -LiteralPath $bak) {
        $src = $bak; Log "Using existing clean backup as source." 'Gray'
    } else {
        Copy-Item -LiteralPath $dll -Destination $bak
        $src = $dll; Log "Backup created: $bak" 'Green'
    }
    $bytes = [byte[]][IO.File]::ReadAllBytes($src)
    $t = Find-Targets $bytes
    if (-not $t.ok) { Log ("FAILED: " + $t.msg) 'Red'; return $false }
    Log ("Analysis: " + $t.msg) 'Gray'
    $caveOff = $t.CaveOff; $caveRva = $t.CaveRva
    $caveFree = $true
    for ($i = 0; $i -lt (2 * $SLOT_LEN); $i++) { if ($bytes[$caveOff + $i] -ne 0) { $caveFree = $false; break } }
    if (-not $caveFree) { Log 'FAILED: cave area not empty' 'Red'; return $false }
    $slot = 0
    foreach ($tg in ($t.Targets | Where-Object { $_.State -eq 'clean' })) {
        $cRva = $caveRva + $slot * $SLOT_LEN
        $sbytes = Build-Slot $tg.Rva $cRva $hi $lo
        $jmp    = Build-Jmp $tg.Rva $cRva
        [Array]::Copy($sbytes, 0, $bytes, $caveOff + $slot * $SLOT_LEN, $SLOT_LEN)
        [Array]::Copy($jmp, 0, $bytes, $tg.Off, 5)
        Log ("patched {0} @0x{1:X}" -f $tg.Cls, $tg.Off) 'Gray'
        $slot++
    }
    if ($slot -eq 0) { Log 'FAILED: nothing to patch (already patched?)' 'Red'; return $false }
    # verify
    $bytes2 = [byte[]][IO.File]::ReadAllBytes($src)
    $slot2 = 0
    foreach ($tg in ($t.Targets | Where-Object { $_.State -eq 'clean' })) {
        $cRva = $caveRva + $slot2 * $SLOT_LEN
        $exp = Build-Slot $tg.Rva $cRva $hi $lo
        for ($i = 0; $i -lt $SLOT_LEN; $i++) {
            if ($bytes[$caveOff + $slot2 * $SLOT_LEN + $i] -ne $exp[$i]) { Log 'FAILED: verify slot' 'Red'; return $false } }
        $slot2++
    }
    try { [IO.File]::WriteAllBytes($dll, $bytes) }
    catch { Log ("FAILED to write (BlueStacks running? quit it): " + $_.Exception.Message) 'Red'; return $false }
    Log ("PATCHED!  high-FPS slider max: {0}   normal mode: {1}" -f $hi, $lo) 'Green'
    Log "Start BlueStacks -> Settings -> Performance -> High frame rate ON ->" 'Gray'
    Log ("slider to {0} -> Save -> restart BlueStacks." -f $hi) 'Gray'
    return $true
}

function Do-Restore([string]$dll) {
    $bak = "$dll.bak"
    if (-not (Test-Path -LiteralPath $bak)) { Log "No backup found: $bak" 'Red'; return $false }
    Copy-Item -LiteralPath $bak -Destination $dll -Force
    Log "ORIGINAL RESTORED. Start BlueStacks normally." 'Green'
    return $true
}

function Find-Dlls {
    $found = @()
    foreach ($root in @($env:ProgramFiles, ${env:ProgramFiles(x86)})) {
        if ($root) {
            $found += Get-Item -ErrorAction SilentlyContinue (Join-Path $root 'BlueStacks*\Qt5QuickTemplates2.dll')
            $found += Get-Item -ErrorAction SilentlyContinue (Join-Path $root 'BlueStacks*\Qt6QuickTemplates2.dll')
        }
    }
    return @($found | Sort-Object FullName -Unique)
}
