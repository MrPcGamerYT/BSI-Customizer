cmake_minimum_required(VERSION 3.21)
project(BsiDllStartupTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Test redistribution without an installed compiler/VC runtime/tool interpreter.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(dll-startup-test test.cpp ../../src/core/DllStartupImport.cpp)
target_include_directories(dll-startup-test PRIVATE ../../src)
if(MSVC)
    target_compile_definitions(dll-startup-test PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(dll-startup-test PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(dll-startup-test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
option(BSI_DLLSTARTUP_FILE_TESTS "Build native QtCore apply/restore fault-injection tests" OFF)
if(BSI_DLLSTARTUP_FILE_TESTS)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
    add_executable(dll-startup-file-test file_test.cpp ../../src/core/DllStartup.cpp ../../src/core/DllStartupImport.cpp)
    target_include_directories(dll-startup-file-test PRIVATE ../../src)
    target_compile_definitions(dll-startup-file-test PRIVATE BSI_DLLSTARTUP_TESTING BSI_DLLSTARTUP_STANDALONE_MAIN)
    target_link_libraries(dll-startup-file-test PRIVATE Qt6::Core)
    if(MSVC)
        set_property(TARGET dll-startup-file-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        target_compile_options(dll-startup-file-test PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(dll-startup-file-test PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME native-file-transactions COMMAND dll-startup-file-test)
endif()
if(WIN32)
    add_executable(startup-host host.cpp)
    add_library(startup-no-exports SHARED payload.cpp)
    add_library(startup-with-export SHARED payload.cpp)
    target_compile_definitions(startup-with-export PRIVATE BSI_PROBE_EXPORT)
    add_library(startup-declines SHARED payload.cpp)
    target_compile_definitions(startup-declines PRIVATE BSI_PROBE_DECLINE BSI_PROBE_EXPORT)
    add_test(NAME windows-normal-startup COMMAND dll-startup-test $<TARGET_FILE:startup-host> $<TARGET_FILE:startup-no-exports> $<TARGET_FILE:startup-with-export> $<TARGET_FILE:startup-declines>)
else()
    add_test(NAME data-only-guards COMMAND dll-startup-test)
endif()
