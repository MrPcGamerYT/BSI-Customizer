cmake_minimum_required(VERSION 3.20)
project(BsiSuppliedPotatoProbe LANGUAGES CXX)
# Deliberately standalone: no Qt, emulator, supplied binary or default CTest.
# Only the explicitly opted-in offline workflow stages/runs an original DLL.
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build this original test host with x64 MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(kind control dinput version)
    add_executable(potato-probe-${kind} probe.cpp)
    target_compile_definitions(potato-probe-${kind} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(potato-probe-${kind} PRIVATE /W4 /WX /utf-8)
    target_include_directories(potato-probe-${kind} PRIVATE ../../src/shaders)
    target_link_libraries(potato-probe-${kind} PRIVATE opengl32 gdi32 user32)
    if(NOT kind STREQUAL "control")
        if(kind STREQUAL "dinput")
            set(anchor proxy_DirectInput8Create)
            target_compile_definitions(potato-probe-${kind} PRIVATE BSI_IMPORT_DINPUT)
        else()
            set(anchor proxy_VerQueryValueW)
            target_compile_definitions(potato-probe-${kind} PRIVATE BSI_IMPORT_VERSION)
        endif()
        set(def "${CMAKE_CURRENT_BINARY_DIR}/${kind}.def")
        set(lib "${CMAKE_CURRENT_BINARY_DIR}/${kind}-original-import.lib")
        file(WRITE "${def}" "LIBRARY BsiPotatoOriginal.dll\nEXPORTS\n${anchor}\n")
        add_custom_command(OUTPUT "${lib}"
            COMMAND "${CMAKE_AR}" /nologo /machine:x64 "/def:${def}" "/out:${lib}"
            DEPENDS "${def}" VERBATIM)
        add_custom_target(potato-import-${kind} DEPENDS "${lib}")
        add_dependencies(potato-probe-${kind} potato-import-${kind})
        target_link_libraries(potato-probe-${kind} PRIVATE "${lib}")
    endif()
endforeach()
